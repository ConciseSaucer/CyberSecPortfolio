import json
from flask_socketio import SocketIO, emit

def stream_cowrie_logs():
    """Asynchronous gevent parser for live JSON logs."""
    with open('/var/log/cowrie.json', 'r') as f:
        f.seek(0, 2) # Tail file
        while True:
            line = f.readline()
            if not line:
                socketio.sleep(0.1); continue
            payload = json.loads(line)
            if payload['eventid'] == 'cowrie.session.connect':
                socketio.emit('new_attack', payload)
